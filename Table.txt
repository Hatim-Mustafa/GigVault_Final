SUPABASE DB CREATION

CREATE TABLE Users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    phone_number VARCHAR(15),
    role VARCHAR(20) NOT NULL CHECK (role IN ('Musician', 'Venue_Owner', 'Admin')),
    profile_picture_url TEXT,
    bio TEXT,
    city VARCHAR(100),
    zip_code VARCHAR(10),
    account_created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE
);

CREATE TABLE Bands (
    band_id SERIAL PRIMARY KEY,
    band_name VARCHAR(150) NOT NULL,
    leader_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    genre VARCHAR(100) NOT NULL,
    bio TEXT,
    profile_picture_url TEXT,
    member_count INTEGER DEFAULT 1,
    city VARCHAR(100),
    zip_code VARCHAR(10),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE
);

CREATE TABLE Band_Members (
    member_id SERIAL PRIMARY KEY,
    band_id INTEGER NOT NULL REFERENCES Bands(band_id) ON DELETE CASCADE,
    user_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    instrument VARCHAR(100) NOT NULL,
    role_in_band VARCHAR(50),
    joined_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(band_id, user_id)
);

CREATE TABLE Gig_Listings (
    gig_id SERIAL PRIMARY KEY,
    venue_owner_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    gig_title VARCHAR(200) NOT NULL,
    description TEXT,
    genre_required VARCHAR(100) NOT NULL,
    performance_date DATE NOT NULL,
    performance_time TIME NOT NULL,
    venue_name VARCHAR(150) NOT NULL,
    location_city VARCHAR(100) NOT NULL,
    location_zip_code VARCHAR(10),
    duration_hours DECIMAL(3,1),
    offered_pay DECIMAL(10,2) NOT NULL,
    payment_status VARCHAR(20) DEFAULT 'Pending' CHECK (payment_status IN ('Pending', 'Paid', 'Partially_Paid')),
    gig_status VARCHAR(20) DEFAULT 'Open' CHECK (gig_status IN ('Open', 'Filled', 'Completed', 'Cancelled')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Applications (
    application_id SERIAL PRIMARY KEY,
    gig_id INTEGER NOT NULL REFERENCES Gig_Listings(gig_id) ON DELETE CASCADE,
    band_id INTEGER NOT NULL REFERENCES Bands(band_id) ON DELETE CASCADE,
    application_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    application_status VARCHAR(20) DEFAULT 'Pending' CHECK (application_status IN ('Pending', 'Accepted', 'Rejected', 'Withdrawn')),
    cover_letter TEXT,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Bookings_Contracts (
    booking_id SERIAL PRIMARY KEY,
    gig_id INTEGER NOT NULL REFERENCES Gig_Listings(gig_id) ON DELETE CASCADE,
    band_id INTEGER NOT NULL REFERENCES Bands(band_id) ON DELETE CASCADE,
    venue_owner_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    contract_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    agreed_fee DECIMAL(10,2) NOT NULL,
    deposit_amount DECIMAL(10,2),
    deposit_status VARCHAR(20) DEFAULT 'Pending' CHECK (deposit_status IN ('Pending', 'Paid')),
    contract_status VARCHAR(20) DEFAULT 'Active' CHECK (contract_status IN ('Active', 'Completed', 'Cancelled')),
    performance_date DATE NOT NULL,
    performance_time TIME NOT NULL,
    contract_terms TEXT,
    signed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP
);

CREATE TABLE Availability_Calendar (
    availability_id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    busy_date DATE NOT NULL,
    busy_time_start TIME,
    busy_time_end TIME,
    reason VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, busy_date)
);

CREATE TABLE Payments (
    payment_id SERIAL PRIMARY KEY,
    booking_id INTEGER NOT NULL REFERENCES Bookings_Contracts(booking_id) ON DELETE CASCADE,
    band_id INTEGER NOT NULL REFERENCES Bands(band_id) ON DELETE CASCADE,
    venue_owner_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    amount DECIMAL(10,2) NOT NULL,
    payment_type VARCHAR(20) NOT NULL CHECK (payment_type IN ('Deposit', 'Final')),
    payment_status VARCHAR(20) DEFAULT 'Pending' CHECK (payment_status IN ('Pending', 'Completed', 'Failed', 'Refunded')),
    payment_date TIMESTAMP,
    payment_method VARCHAR(50),
    transaction_id VARCHAR(100) UNIQUE,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Recruitment_Ads (
    recruitment_id SERIAL PRIMARY KEY,
    band_id INTEGER NOT NULL REFERENCES Bands(band_id) ON DELETE CASCADE,
    posted_by_user_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    title VARCHAR(200) NOT NULL,
    description TEXT,
    instruments_needed VARCHAR(255) NOT NULL,
    genre VARCHAR(100) NOT NULL,
    experience_level VARCHAR(50) CHECK (experience_level IN ('Beginner', 'Intermediate', 'Advanced', 'Professional')),
    city VARCHAR(100),
    zip_code VARCHAR(10),
    ad_status VARCHAR(20) DEFAULT 'Active' CHECK (ad_status IN ('Active', 'Filled', 'Closed')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Reviews_Disputes (
    review_id SERIAL PRIMARY KEY,
    booking_id INTEGER NOT NULL REFERENCES Bookings_Contracts(booking_id) ON DELETE CASCADE,
    reviewer_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    reviewee_id INTEGER NOT NULL REFERENCES Users(user_id) ON DELETE CASCADE,
    rating INTEGER CHECK (rating >= 1 AND rating <= 5),
    review_text TEXT,
    review_type VARCHAR(20) DEFAULT 'Review' CHECK (review_type IN ('Review', 'Dispute')),
    dispute_reason VARCHAR(255),
    is_flagged BOOLEAN DEFAULT FALSE,
    admin_notes TEXT,
    status VARCHAR(20) DEFAULT 'Open' CHECK (status IN ('Open', 'Resolved', 'Pending_Review')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    resolved_at TIMESTAMP
);

CREATE TABLE Setlists (
    setlist_id SERIAL PRIMARY KEY,
    band_id INTEGER NOT NULL REFERENCES Bands(band_id) ON DELETE CASCADE,
    gig_id INTEGER REFERENCES Gig_Listings(gig_id) ON DELETE SET NULL,
    setlist_name VARCHAR(200) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Setlist_Songs (
    song_id SERIAL PRIMARY KEY,
    setlist_id INTEGER NOT NULL REFERENCES Setlists(setlist_id) ON DELETE CASCADE,
    song_title VARCHAR(200) NOT NULL,
    artist_name VARCHAR(150),
    duration_minutes DECIMAL(4,2),
    song_order INTEGER NOT NULL,
    genre VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_users_email ON Users(email);
CREATE INDEX idx_users_role ON Users(role);
CREATE INDEX idx_bands_leader_id ON Bands(leader_id);
CREATE INDEX idx_bands_genre ON Bands(genre);
CREATE INDEX idx_gig_listings_venue_owner_id ON Gig_Listings(venue_owner_id);
CREATE INDEX idx_gig_listings_genre ON Gig_Listings(genre_required);
CREATE INDEX idx_gig_listings_status ON Gig_Listings(gig_status);
CREATE INDEX idx_gig_listings_date ON Gig_Listings(performance_date);
CREATE INDEX idx_applications_gig_id ON Applications(gig_id);
CREATE INDEX idx_applications_band_id ON Applications(band_id);
CREATE INDEX idx_applications_status ON Applications(application_status);
CREATE INDEX idx_bookings_contracts_band_id ON Bookings_Contracts(band_id);
CREATE INDEX idx_bookings_contracts_gig_id ON Bookings_Contracts(gig_id);
CREATE INDEX idx_bookings_contracts_status ON Bookings_Contracts(contract_status);
CREATE INDEX idx_availability_calendar_user_id ON Availability_Calendar(user_id);
CREATE INDEX idx_availability_calendar_date ON Availability_Calendar(busy_date);
CREATE INDEX idx_payments_booking_id ON Payments(booking_id);
CREATE INDEX idx_payments_status ON Payments(payment_status);
CREATE INDEX idx_recruitment_ads_band_id ON Recruitment_Ads(band_id);
CREATE INDEX idx_recruitment_ads_status ON Recruitment_Ads(ad_status);
CREATE INDEX idx_reviews_disputes_booking_id ON Reviews_Disputes(booking_id);
CREATE INDEX idx_reviews_disputes_status ON Reviews_Disputes(status);
CREATE INDEX idx_reviews_disputes_flagged ON Reviews_Disputes(is_flagged);
CREATE INDEX idx_setlists_band_id ON Setlists(band_id);
CREATE INDEX idx_setlist_songs_setlist_id ON Setlist_Songs(setlist_id);
